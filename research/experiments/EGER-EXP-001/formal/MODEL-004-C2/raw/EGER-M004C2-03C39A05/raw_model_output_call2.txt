```sdc
set sdc_version 2.2
set_units -time ns -capacitance pF

create_clock -name clk -period 10.0 [get_ports clk]
create_generated_clock -name clk_div2 -divide_by 2 -source [get_pins div_reg/Q] [get_ports clk_div2]

set_clock_groups -asynchronous -group {clk} -group {clk_div2}

set_clock_latency -source 0.5 [get_clocks clk]
set_clock_transition 0.1 [get_clocks {clk clk_div2}]
set_clock_jitter 0.05 [get_clocks {clk clk_div2}]

set_input_delay -clock clk -max 2.0 [all_inputs]
set_input_delay -clock clk -min 0.5 [all_inputs]
set_input_transition 0.2 [all_inputs]

set_output_delay -clock clk -max 3.0 [all_outputs]
set_output_delay -clock clk -min 0.5 [all_outputs]
set_load 10.0 [all_outputs]

set_max_fanout 20 [all_inputs]
set_max_transition 0.2 [all_nets]
set_max_capacitance 50.0 [all_nets]

set_driving_cell -lib_cell BUF_X1 [all_inputs]

set_operating_conditions -max_library WCCOM -min_library BCCOM
set_timing_derate -cell_delay -max 1.1 -min 0.9
set_timing_derate -net_delay -max 1.1 -min 0.9

set_case_analysis 0 [get_ports scan_en]
set_case_analysis 0 [get_ports test_mode]
set_ideal_network [get_ports {scan_en test_mode reset_n}]

set_dont_use [get_cells -hier *WEAK*]
set_max_area 100000

set_propagated_clock [all_clocks]
```
